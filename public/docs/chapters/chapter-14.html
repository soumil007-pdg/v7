Deployment Options

Vercel (Recommended)

npm install -g vercel
vercel

Docker

FROM node:18
WORKDIR /app
COPY package.json .
RUN npm install
COPY . .
RUN npm run build
EXPOSE 3000
CMD ["npm", "start"]
