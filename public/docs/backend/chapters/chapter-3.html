🔌 Chapter 3: Next.js API Routes Deep Dive

Next.js API Routes are the easiest way to build a backend in Next.js. A route handler is just a function that receives a request and sends a response.

Creating an API Route

File path becomes the endpoint:

app/api/auth/login/route.js
      ↓↓↓
POST http://localhost:3000/api/auth/login

Route Handler Basics

// app/api/hello/route.js
export async function GET(request) {
  return Response.json({ message: 'Hello!' });
}

// URL: http://localhost:3000/api/hello
// Response: { "message": "Hello!" }

HTTP Methods

Method Purpose When to Use
GET Retrieve data Fetching user info, getting list of items
POST Create/send data Login, signup, creating a post
PUT Update entire resource Updating user profile
DELETE Delete data Deleting a post, account

Handling POST Requests

// app/api/auth/login/route.js
export async function POST(request) {
  // Get the request body (JSON)
  const { email, password } = await request.json();

  // Process the data
  console.log('User trying to login:', email);

  // Return a response
  return Response.json({
    success: true,
    message: 'Login successful'
  });
}

Extracting Query Parameters

// app/api/search/route.js
export async function GET(request) {
  // Get query params: /api/search?q=javascript
  const url = new URL(request.url);
  const query = url.searchParams.get('q');

  console.log('Search for:', query);

  return Response.json({
    results: ['JavaScript basics', 'Advanced JS patterns']
  });
}

Getting the Body from POST

export async function POST(request) {
  try {
    const body = await request.json();
    console.log('Received:', body);
    return Response.json({ received: true });
  } catch (error) {
    return Response.json(
      { error: 'Invalid JSON' },
      { status: 400 }
    );
  }
}

Setting Response Status

export async function POST(request) {
  const { email, password } = await request.json();

  // Check if valid
  if (!email || !password) {
    // Return 400 Bad Request
    return Response.json(
      { error: 'Email and password required' },
      { status: 400 }
    );
  }

  // Success - return 200 (default)
  return Response.json({ success: true });
}

Setting Headers

export async function GET() {
  return Response.json(
    { message: 'Hello' },
    {
      status: 200,
      headers: {
        'Content-Type': 'application/json',
        'X-Custom-Header': 'value'
      }
    }
  );
}

Real Example: Login Route

// app/api/auth/login/route.js
import { users, sessions } from '@/lib/authStore';
import crypto from 'crypto';

export async function POST(request) {
  try {
    const { email, password } = await request.json();

    // Find user
    const user = Array.from(users.values()).find(u => u.email === email);

    if (!user) {
      return Response.json(
        { error: 'User not found' },
        { status: 404 }
      );
    }

    // Create token
    const token = crypto.randomUUID();
    sessions.set(token, {
      userId: user.id,
      createdAt: Date.now()
    });

    return Response.json({
      success: true,
      token: token
    });
  } catch (error) {
    return Response.json(
      { error: 'Server error' },
      { status: 500 }
    );
  }
}

Key Concepts

  • Route Handler: Function that handles a request
  • Request: Object containing method, URL, headers, body
  • Response: Object to send back to client
  • Status Code: Number indicating success/failure (200, 400, 500)
  • Headers: Metadata about the response
  • Body: The actual data being sent/received

Common Status Codes

  • 200 - Success
  • 201 - Created
  • 400 - Bad Request (invalid data)
  • 401 - Unauthorized (not logged in)
  • 404 - Not Found
  • 500 - Server Error
