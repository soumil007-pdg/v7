🔄 Chapter 9: Middleware & Request Pipeline

Middleware is code that runs BEFORE your route handlers. It's perfect for authentication, logging, and request processing.

What is Middleware?

Middleware intercepts requests before they reach your route handlers:

Request comes in
  ↓
Middleware 1 runs (check locale)
  ↓
Middleware 2 runs (check auth)
  ↓
Route handler runs (/api/chat)
  ↓
Response sent back

ADVOCAT-Easy's Middleware

// middleware.js (runs on every request)
import { NextResponse } from 'next/server';

export function middleware(request) {
  // Get the pathname
  const pathname = request.nextUrl.pathname;

  // Check locale routing
  const locales = ['en', 'hi', 'mr', 'te'];
  const hasLocale = locales.some(locale => pathname.startsWith(`/${locale}`));

  if (!hasLocale) {
    // Redirect to English if no locale
    return NextResponse.redirect(new URL('/en' + pathname, request.url));
  }

  return NextResponse.next();
}

// config: Tell Next.js which paths use middleware
export const config = {
  matcher: ['/((?!_next).*)']
};

Custom Middleware for Auth

// middleware.js
export function middleware(request) {
  const pathname = request.nextUrl.pathname;

  // Routes that need authentication
  const protectedPaths = [
    '/api/chat',
    '/api/case-advisor',
    '/api/user'
  ];

  const needsAuth = protectedPaths.some(path => pathname.startsWith(path));

  if (needsAuth) {
    const authHeader = request.headers.get('Authorization');
    const token = authHeader?.replace('Bearer ', '');

    if (!token) {
      return new Response(
        JSON.stringify({ error: 'No token' }),
        { status: 401, headers: { 'Content-Type': 'application/json' } }
      );
    }

    // Continue if token exists
  }

  return NextResponse.next();
}

Request Logging Middleware

// middleware.js
export function middleware(request) {
  const start = Date.now();
  const pathname = request.nextUrl.pathname;
  const method = request.method;

  console.log(`→ ${method} ${pathname}`);

  const response = NextResponse.next();

  const duration = Date.now() - start;
  console.log(`← ${method} ${pathname} (${duration}ms)`);

  return response;
}

Adding Custom Headers

// middleware.js
export function middleware(request) {
  const response = NextResponse.next();

  // Add security headers
  response.headers.set('X-Content-Type-Options', 'nosniff');
  response.headers.set('X-Frame-Options', 'DENY');
  response.headers.set('X-XSS-Protection', '1; mode=block');

  // Add custom header
  response.headers.set('X-Custom-Middleware', 'processed');

  return response;
}

Middleware in Route Handlers

You can also create middleware-like functions in your routes:

// app/api/chat/route.js
function authenticateRequest(request) {
  const authHeader = request.headers.get('Authorization');
  const token = authHeader?.replace('Bearer ', '');

  if (!token) {
    throw new Error('No token');
  }

  return token;
}

function validateRequestBody(body) {
  if (!body.message) {
    throw new Error('Message required');
  }

  return body;
}

export async function POST(request) {
  try {
    const token = authenticateRequest(request);
    const body = await request.json();
    validateRequestBody(body);

    // Process chat...
    return Response.json({ success: true });

  } catch (error) {
    return Response.json(
      { error: error.message },
      { status: 400 }
    );
  }
}

Middleware Chain

Multiple middleware can run in sequence:

// lib/middleware.js
export const middlewares = {
  // Check authentication
  async authenticate(request) {
    const token = request.headers.get('Authorization')?.replace('Bearer ', '');
    if (!token) throw new Error('No token');
    return token;
  },

  // Check rate limit
  async rateLimit(request, token) {
    const ip = request.headers.get('x-forwarded-for');
    if (isRateLimited(ip)) throw new Error('Rate limited');
  },

  // Log request
  async log(request) {
    console.log(`${request.method} ${request.nextUrl.pathname}`);
  }
};

// Usage in route
export async function POST(request) {
  try {
    await middlewares.log(request);
    const token = await middlewares.authenticate(request);
    await middlewares.rateLimit(request, token);

    // Continue...

  } catch (error) {
    return Response.json({ error: error.message }, { status: 400 });
  }
}

Global Middleware vs Route Middleware

Global (middleware.js) Route (in handler)
Runs for ALL requests Runs for specific route
Can't access request body Can access body
Good for auth, logging Good for validation
Centralized control Route-specific logic

CORS Middleware

// Allow frontend to call your API
export async function middleware(request) {
  // Handle preflight requests
  if (request.method === 'OPTIONS') {
    return new Response(null, {
      headers: {
        'Access-Control-Allow-Origin': 'http://localhost:3000',
        'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE',
        'Access-Control-Allow-Headers': 'Content-Type, Authorization'
      }
    });
  }

  return NextResponse.next();
}

Error Handling in Middleware

export function middleware(request) {
  try {
    // Validate request
    if (request.method !== 'GET' && request.method !== 'POST') {
      return new Response('Method not allowed', { status: 405 });
    }

    return NextResponse.next();

  } catch (error) {
    console.error('Middleware error:', error);
    return new Response('Internal error', { status: 500 });
  }
}

Key Concepts

  • Middleware: Code that runs before route handlers
  • Global middleware: middleware.js runs for all requests
  • Route middleware: Custom functions in handlers
  • Request pipeline: Order of operations
  • Intercepting: Examining and modifying requests
