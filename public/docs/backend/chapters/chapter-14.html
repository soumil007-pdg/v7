⏱️ Chapter 14: Rate Limiting & Throttling

Rate limiting prevents abuse by restricting how many requests a user can make. Let's implement it.

Why Rate Limiting?

  • Prevent abuse: Stop attackers from hammering your API
  • Fair usage: Ensure one user doesn't monopolize resources
  • Cost control: Limit expensive API calls (Gemini, Apify)
  • Server stability: Prevent overwhelming the server

Simple Rate Limiting

// lib/rateLimit.js
const requestCounts = new Map();

export function isRateLimited(identifier, limit = 10, windowMs = 60000) {
  const now = Date.now();
  const key = identifier;

  if (!requestCounts.has(key)) {
    requestCounts.set(key, []);
  }

  const requests = requestCounts.get(key);

  // Remove old requests (older than window)
  const recentRequests = requests.filter(time => now - time < windowMs);

  if (recentRequests.length >= limit) {
    return true;  // Rate limited
  }

  // Add this request
  recentRequests.push(now);
  requestCounts.set(key, recentRequests);

  return false;  // Not rate limited
}

Using Rate Limiting in Routes

// app/api/auth/login/route.js
import { isRateLimited } from '@/lib/rateLimit';

export async function POST(request) {
  // Get client identifier (IP address)
  const ip = request.headers.get('x-forwarded-for') || 'unknown';

  // Rate limit: 5 attempts per minute
  if (isRateLimited(ip, 5, 60000)) {
    return Response.json(
      { error: 'Too many login attempts. Try again in 1 minute.' },
      { status: 429 }  // Too Many Requests
    );
  }

  // Continue with login...
  const { email, password } = await request.json();
  // ... rest of login logic
}

Per-User Rate Limiting

Limit based on user ID instead of IP (for authenticated endpoints):

// app/api/chat/route.js
import { isRateLimited } from '@/lib/rateLimit';

export async function POST(request) {
  const token = request.headers.get('Authorization')?.replace('Bearer ', '');
  const session = sessions.get(token);

  if (!session) {
    return Response.json({ error: 'Unauthorized' }, { status: 401 });
  }

  const userId = session.userId;

  // Rate limit: 50 messages per hour
  if (isRateLimited(`chat-${userId}`, 50, 3600000)) {
    return Response.json(
      { error: 'Message limit reached. Try again later.' },
      { status: 429 }
    );
  }

  // Continue with chat...
}

Response with Retry Info

// Return when rate limited
export function createRateLimitError(resetAt) {
  const now = Date.now();
  const secondsUntilReset = Math.ceil((resetAt - now) / 1000);

  return Response.json(
    {
      error: 'Rate limit exceeded',
      retryAfter: secondsUntilReset
    },
    {
      status: 429,
      headers: {
        'Retry-After': secondsUntilReset.toString()
      }
    }
  );
}

Advanced Rate Limiting

Track remaining requests and reset time:

// lib/rateLimit.js (advanced)
export function checkRateLimit(identifier, limit = 10, windowMs = 60000) {
  const now = Date.now();

  if (!requestCounts.has(identifier)) {
    requestCounts.set(identifier, { count: 0, resetAt: now + windowMs });
  }

  const record = requestCounts.get(identifier);

  // Reset window?
  if (now > record.resetAt) {
    record.count = 0;
    record.resetAt = now + windowMs;
  }

  // Check limit
  const isLimited = record.count >= limit;

  if (!isLimited) {
    record.count += 1;
  }

  const remaining = Math.max(0, limit - record.count);
  const resetIn = record.resetAt - now;

  return {
    isLimited,
    remaining,
    resetIn,
    resetAt: record.resetAt
  };
}

// Usage
export async function POST(request) {
  const userId = 'user-123';
  const { isLimited, remaining, resetIn } = checkRateLimit(
    `chat-${userId}`,
    50,
    3600000
  );

  if (isLimited) {
    return Response.json(
      {
        error: 'Rate limit exceeded',
        resetIn: Math.ceil(resetIn / 1000)
      },
      {
        status: 429,
        headers: {
          'X-RateLimit-Remaining': '0',
          'X-RateLimit-Reset': (Date.now() + resetIn).toString()
        }
      }
    );
  }

  // Return response with rate limit info
  return Response.json(
    { success: true, data: '...' },
    {
      status: 200,
      headers: {
        'X-RateLimit-Remaining': remaining.toString(),
        'X-RateLimit-Reset': (Date.now() + resetIn).toString()
      }
    }
  );
}

Cleanup Expired Records

Periodically clean up old rate limit records:

// Clean every hour
setInterval(() => {
  const now = Date.now();

  for (const [key, record] of requestCounts.entries()) {
    if (now > record.resetAt) {
      requestCounts.delete(key);
    }
  }
}, 3600000);  // 1 hour

Rate Limit Headers

Standard HTTP headers for rate limiting:

return Response.json(
  { data: '...' },
  {
    headers: {
      'X-RateLimit-Limit': '50',           // Total limit
      'X-RateLimit-Remaining': '45',       // Remaining requests
      'X-RateLimit-Reset': '1234567890'    // Reset timestamp
    }
  }
);

Sliding Window vs Fixed Window

Fixed Window Sliding Window
Reset at fixed times (0:00, 1:00, 2:00) Reset 60 seconds after each request
Simpler to implement More fair to users
Can allow spike at boundary Prevents spikes

Key Concepts

  • Rate limiting: Restrict request frequency
  • Throttling: Slow down requests
  • 429: Too Many Requests status code
  • Window: Time period for counting requests
  • Identifier: IP address or user ID
  • Retry-After: Header telling client when to retry
