📨 Chapter 11: Request/Response Handling

Every API interaction involves a request (from client) and a response (from server). Let's master handling both.

Anatomy of a Request

export async function POST(request) {
  // Method: GET, POST, PUT, DELETE
  console.log(request.method);  // "POST"

  // URL and query params
  const url = new URL(request.url);
  console.log(url.pathname);  // "/api/auth/login"
  console.log(url.searchParams.get('page'));  // Query param

  // Headers (metadata)
  const contentType = request.headers.get('Content-Type');
  const token = request.headers.get('Authorization');

  // Body (the actual data)
  const body = await request.json();
  console.log(body);  // { email: "...", password: "..." }
}

Real-World Example: Complete Request/Response

// app/api/auth/login/route.js
import { users, sessions } from '@/lib/authStore';
import bcrypt from 'bcryptjs';
import crypto from 'crypto';

export async function POST(request) {
  try {
    // 1. EXTRACT: Get request data
    const { email, password } = await request.json();

    // 2. VALIDATE: Check data is correct
    if (!email || !password) {
      return Response.json(
        { error: 'Email and password required' },
        { status: 400 }
      );
    }

    // 3. PROCESS: Find user, verify password
    const user = Array.from(users.values()).find(u => u.email === email);

    if (!user) {
      return Response.json(
        { error: 'Invalid credentials' },
        { status: 401 }
      );
    }

    const isValid = await bcrypt.compare(password, user.password);

    if (!isValid) {
      return Response.json(
        { error: 'Invalid credentials' },
        { status: 401 }
      );
    }

    // 4. CREATE: Generate session
    const token = crypto.randomUUID();
    sessions.set(token, {
      userId: user.id,
      createdAt: Date.now()
    });

    // 5. RESPOND: Send success response
    return Response.json(
      {
        success: true,
        token: token,
        user: {
          id: user.id,
          email: user.email
        }
      },
      { status: 200 }
    );

  } catch (error) {
    // 6. ERROR: Handle unexpected errors
    console.error('Login error:', error);
    return Response.json(
      { error: 'Something went wrong' },
      { status: 500 }
    );
  }
}

Streaming Responses

For large responses, stream data gradually:

export async function GET(request) {
  // For AI chat streaming (chunks of text as they arrive)

    async start(controller) {

  });

Request Validation Pattern

async function validateLoginRequest(request) {
  try {
    const body = await request.json();

    const { email, password } = body;

    if (!email) throw new Error('Email required');
    if (!password) throw new Error('Password required');

    if (typeof email !== 'string') throw new Error('Email must be string');
    if (typeof password !== 'string') throw new Error('Password must be string');

    return { valid: true, data: { email, password } };

  } catch (error) {
    return {
      valid: false,
      error: error.message
    };
  }
}

export async function POST(request) {
  const { valid, error, data } = await validateLoginRequest(request);

  if (!valid) {
    return Response.json({ error }, { status: 400 });
  }

  // Continue with validated data...
}

Response Caching

// Tell client to cache for 1 hour
export async function GET(request) {
  return Response.json(
    { data: 'cacheable' },
    {
      headers: {
        'Cache-Control': 'public, max-age=3600'
      }
    }
  );
}

// Don't cache (sensitive data)
export async function GET(request) {
  return Response.json(
    { data: 'user-sensitive' },
    {
      headers: {
        'Cache-Control': 'no-cache, no-store, must-revalidate'
      }
    }
  );
}

Key Concepts

  • Request: Data sent from client to server
  • Response: Data sent from server back to client
  • Headers: Metadata in request/response
  • Body: The actual data (JSON, text, binary)
  • Status code: Indicates success or failure
  • Content-Type: Describes what kind of data
