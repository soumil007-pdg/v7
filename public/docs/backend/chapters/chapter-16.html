🤖 Chapter 16: Google Gemini AI Integration

ADVOCAT-Easy uses Google's Gemini AI to provide legal guidance. Let's learn how to integrate and use it.

What is Gemini?

Gemini is Google's AI model that understands text and can generate intelligent responses. Perfect for legal advice!

Setup

# Install Google Generative AI
npm install @google/generative-ai

# Add API key to .env
GEMINI_API_KEY=your-api-key-here

Basic Usage

// app/api/chat/route.js
import { GoogleGenerativeAI } from '@google/generative-ai';

const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY);
const model = genAI.getGenerativeModel({ model: 'gemini-2.5-flash' });

export async function POST(request) {
  const { message } = await request.json();

  const result = await model.generateContent(message);
  const text = result.response.text();

  return Response.json({
    success: true,
    response: text
  });
}

Multi-Turn Conversation

Keep conversation history for context:

// app/api/chat/route.js
export async function POST(request) {
  const { conversationId, message } = await request.json();

  // Get conversation history from database
  const conversation = conversations.get(conversationId);
  const history = conversation?.messages || [];

  // Build conversation history for Gemini
  const contents = history.map(msg => ({
    role: msg.role === 'user' ? 'user' : 'model',
    parts: [{ text: msg.content }]
  }));

  // Add new message
  contents.push({
    role: 'user',
    parts: [{ text: message }]
  });

  // Get response from Gemini
  const chat = model.startChat({ history: contents });
  const result = await chat.sendMessage(message);
  const response = result.response.text();

  // Save to conversation
  if (!conversation) {
    conversations.set(conversationId, {
      id: conversationId,
      userId: userId,
      messages: []
    });
  }

  conversations.get(conversationId).messages.push({
    role: 'user',
    content: message,
    timestamp: Date.now()
  });

  conversations.get(conversationId).messages.push({
    role: 'assistant',
    content: response,
    timestamp: Date.now()
  });

  return Response.json({
    success: true,
    response: response
  });
}

Safety Settings

Configure Gemini to avoid harmful content:

import { HarmCategory, HarmBlockThreshold } from '@google/generative-ai';

const model = genAI.getGenerativeModel({
  model: 'gemini-2.5-flash',
  safetySettings: [
    {
      category: HarmCategory.HARM_CATEGORY_HARASSMENT,
      threshold: HarmBlockThreshold.BLOCK_MEDIUM_AND_ABOVE
    },
    {
      category: HarmCategory.HARM_CATEGORY_HATE_SPEECH,
      threshold: HarmBlockThreshold.BLOCK_MEDIUM_AND_ABOVE
    },
    {
      category: HarmCategory.HARM_CATEGORY_SEXUALLY_EXPLICIT,
      threshold: HarmBlockThreshold.BLOCK_MEDIUM_AND_ABOVE
    },
    {
      category: HarmCategory.HARM_CATEGORY_DANGEROUS_CONTENT,
      threshold: HarmBlockThreshold.BLOCK_MEDIUM_AND_ABOVE
    }
  ]
});

export async function POST(request) {
  try {
    const { message } = await request.json();
    const result = await model.generateContent(message);

    // Check if content was blocked
    if (!result.response.text()) {
      return Response.json(
        { error: 'Response was blocked by safety settings' },
        { status: 400 }
      );
    }

    return Response.json({
      success: true,
      response: result.response.text()
    });

  } catch (error) {
    console.error('Gemini error:', error);
    return Response.json(
      { error: 'Failed to generate response' },
      { status: 500 }
    );
  }
}

Error Handling

export async function POST(request) {
  try {
    const { message } = await request.json();

    if (!message) {
      return Response.json(
        { error: 'Message is required' },
        { status: 400 }
      );
    }

    if (message.length > 5000) {
      return Response.json(
        { error: 'Message is too long (max 5000 characters)' },
        { status: 400 }
      );
    }

    const result = await model.generateContent(message);
    const response = result.response.text();

    if (!response) {
      return Response.json(
        { error: 'No response from AI' },
        { status: 500 }
      );
    }

    return Response.json({ success: true, response });

  } catch (error) {
    console.error('Gemini error:', error.message);

    // Handle rate limiting
    if (error.message.includes('429') || error.message.includes('too many')) {
      return Response.json(
        { error: 'Too many requests. Please try again later.' },
        { status: 429 }
      );
    }

    // Handle API errors
    if (error.message.includes('API') || error.message.includes('auth')) {
      return Response.json(
        { error: 'AI service error. Please try again.' },
        { status: 503 }
      );
    }

    return Response.json(
      { error: 'Failed to process request' },
      { status: 500 }
    );
  }
}

Streaming Responses

Stream Gemini responses word-by-word:

export async function POST(request) {
  const { message } = await request.json();

  const stream = new ReadableStream({
    async start(controller) {
      try {
        const result = await model.generateContentStream(message);

        for await (const chunk of result.stream) {
          const text = chunk.text();
          controller.enqueue(text);
          // Small delay for demo (remove in production)
          await new Promise(r => setTimeout(r, 10));
        }

        controller.close();
      } catch (error) {
        controller.error(error);
      }
    }
  });

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/plain',
      'Transfer-Encoding': 'chunked'
    }
  });
}

Cost Optimization

// Cache responses for similar queries
const responseCache = new Map();

export async function POST(request) {
  const { message } = await request.json();

  // Check cache
  if (responseCache.has(message)) {
    const cached = responseCache.get(message);
    // Check if cache is fresh (1 hour)
    if (Date.now() - cached.timestamp < 3600000) {
      return Response.json({
        success: true,
        response: cached.response,
        cached: true
      });
    }
  }

  // Get fresh response
  const result = await model.generateContent(message);
  const response = result.response.text();

  // Cache it
  responseCache.set(message, {
    response,
    timestamp: Date.now()
  });

  return Response.json({
    success: true,
    response: response
  });
}
