🧪 Chapter 13: API Testing & Debugging

Test your APIs before deploying! We'll use curl, browser DevTools, and logging to debug issues.

Testing with curl

curl is a command-line tool for making HTTP requests:

# Simple GET request
curl http://localhost:3000/api/users

# POST request with data
curl -X POST http://localhost:3000/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"user@example.com","password":"123456"}'

# With authorization header
curl -X POST http://localhost:3000/api/chat \
  -H "Authorization: Bearer token-abc123" \
  -H "Content-Type: application/json" \
  -d '{"message":"Hello"}'

# With query parameters
curl "http://localhost:3000/api/users?page=1&limit=10"

# Show response headers
curl -i http://localhost:3000/api/users

# Show request and response
curl -v http://localhost:3000/api/users

Testing with Browser DevTools

Open DevTools (F12) → Network tab:

1. Open DevTools (F12)
2. Click "Network" tab
3. Make a request (navigate, click button, submit form)
4. Click the request to see:
   - Status code (200, 400, 401, etc.)
   - Request headers (Authorization, etc.)
   - Request body (what was sent)
   - Response headers
   - Response body (what came back)
5. Check Console tab for JavaScript errors

Server-Side Logging

// app/api/auth/login/route.js
export async function POST(request) {
  // Log request details
  console.log('=== LOGIN REQUEST ===');
  console.log('Method:', request.method);
  console.log('URL:', request.url);
  console.log('Headers:', Object.fromEntries(request.headers));

  const { email, password } = await request.json();
  console.log('Email:', email);
  console.log('Password:', password.substring(0, 2) + '***');  // Don't log full password!

  // Log processing
  console.log('Looking for user...');
  const user = Array.from(users.values()).find(u => u.email === email);

  if (!user) {
    console.log('User not found:', email);
    return Response.json({ error: 'Invalid credentials' }, { status: 401 });
  }

  console.log('User found:', user.id);

  // Continue...
  console.log('Login successful');
  return Response.json({ success: true, token: '...' });
}

Debugging with Console

// app/api/chat/route.js
export async function POST(request) {
  try {
    const token = request.headers.get('Authorization');
    console.log('Token:', token);

    const body = await request.json();
    console.log('Request body:', body);

    // Debug state
    console.log('Current sessions:', sessions.size);
    console.log('Current users:', users.size);

    // Debug processing
    const session = sessions.get(token);
    console.log('Session found:', !!session);

    if (session) {
      const user = users.get(session.userId);
      console.log('User:', user.email);
    }

    // Call API
    const response = await callGemini(body.message);
    console.log('Gemini response:', response);

    return Response.json({ success: true });

  } catch (error) {
    console.error('ERROR:', error);
    console.error('Stack:', error.stack);
    return Response.json({ error: error.message }, { status: 500 });
  }
}

Testing Different Scenarios

# Test 1: Login with valid credentials
curl -X POST http://localhost:3000/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"test@example.com","password":"password123"}'
# Expected: 200 with token

# Test 2: Login with invalid email
curl -X POST http://localhost:3000/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"wrong@example.com","password":"password123"}'
# Expected: 401 Unauthorized

# Test 3: Login with missing password
curl -X POST http://localhost:3000/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"test@example.com"}'
# Expected: 400 Bad Request

# Test 4: Use token in protected route
curl -X POST http://localhost:3000/api/chat \
  -H "Authorization: Bearer abc123" \
  -H "Content-Type: application/json" \
  -d '{"message":"Hello"}'
# Expected: 200 Success

# Test 5: Protected route without token
curl -X POST http://localhost:3000/api/chat \
  -H "Content-Type: application/json" \
  -d '{"message":"Hello"}'
# Expected: 401 Unauthorized

Structured Logging

// lib/logger.js
export const logger = {
  info: (action, details) => {
    console.log(`[INFO] ${action}:`, details);
  },
  error: (action, error) => {
    console.error(`[ERROR] ${action}:`, error.message);
  },
  debug: (action, details) => {
    if (process.env.DEBUG) {
      console.log(`[DEBUG] ${action}:`, details);
    }
  }
};

// Usage
import { logger } from '@/lib/logger';

export async function POST(request) {
  logger.info('Login attempt', { email: 'user@example.com' });

  try {
    const result = await authenticate();
    logger.info('Login success', { userId: result.id });
    return Response.json(result);

  } catch (error) {
    logger.error('Login failed', error);
    return Response.json({ error: 'Login failed' }, { status: 500 });
  }
}

Performance Debugging

export async function POST(request) {
  const start = Date.now();

  // Process request
  const body = await request.json();
  const result = await expensiveOperation(body);

  const duration = Date.now() - start;
  console.log(`Request took ${duration}ms`);

  if (duration > 1000) {
    console.warn('Slow request detected!');
  }

  return Response.json(result);
}

Common Issues & Solutions

Issue Solution
401 Unauthorized on protected route Check Authorization header in DevTools Network tab
404 Not Found Check the URL path matches the file path
400 Bad Request Check request body JSON syntax
500 Server Error Check server console for error logs
CORS error in browser Check browser console, add CORS headers

Postman for API Testing

Postman is a GUI for testing APIs:

1. Download Postman (free)
2. Create new request
3. Set method (GET, POST, etc.)
4. Enter URL (http://localhost:3000/api/...)
5. Add headers (Authorization, Content-Type)
6. Add body (JSON)
7. Click Send
8. View response, status code, headers

Key Concepts

  • curl: Command-line API testing
  • DevTools Network: Browser request/response inspection
  • console.log: Print debug info to server console
  • Status code: Indicates success/failure
  • Headers: Authorization, Content-Type, etc.
  • Postman: GUI for API testing
