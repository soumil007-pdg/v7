⚠️ Chapter 8: Error Handling & Validation

Every API endpoint should validate input and handle errors gracefully. Never let bad data crash your server!

Why Validate Input?

  • Security: Prevent injection attacks
  • Data Integrity: Ensure data is correct
  • User Experience: Tell users what went wrong
  • Stability: Prevent crashes

Basic Validation

// app/api/auth/login/route.js
export async function POST(request) {
  const { email, password } = await request.json();

  // Validate email
  if (!email) {
    return Response.json(
      { error: 'Email is required' },
      { status: 400 }
    );
  }

  // Validate email format
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!emailRegex.test(email)) {
    return Response.json(
      { error: 'Email is invalid' },
      { status: 400 }
    );
  }

  // Validate password
  if (!password) {
    return Response.json(
      { error: 'Password is required' },
      { status: 400 }
    );
  }

  if (password.length < 6) {
    return Response.json(
      { error: 'Password must be at least 6 characters' },
      { status: 400 }
    );
  }

  // Proceed with login...
}

Try/Catch for Error Handling

export async function POST(request) {
  try {
    const { email, password } = await request.json();

    // Validation
    if (!email || !password) {
      return Response.json(
        { error: 'Missing fields' },
        { status: 400 }
      );
    }

    // Process login
    const user = findUser(email);
    const valid = await bcrypt.compare(password, user.password);

    if (!valid) {
      return Response.json(
        { error: 'Invalid credentials' },
        { status: 401 }
      );
    }

    return Response.json({ success: true, token: createToken() });

  } catch (error) {
    console.error('Login error:', error);

    // Don't expose internal errors to users
    return Response.json(
      { error: 'Something went wrong' },
      { status: 500 }
    );
  }
}

Validation Helpers

// lib/validation.js
export function validateEmail(email) {
  const regex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return regex.test(email);
}

export function validatePassword(password) {
  return password && password.length >= 6;
}

export function validateRequest(data, schema) {
  const errors = [];

  if (!data.email) errors.push('Email required');
  else if (!validateEmail(data.email)) errors.push('Invalid email');

  if (!data.password) errors.push('Password required');
  else if (!validatePassword(data.password)) errors.push('Password too short');

  return {
    valid: errors.length === 0,
    errors
  };
}

// Usage in route
import { validateRequest } from '@/lib/validation';

export async function POST(request) {
  const body = await request.json();
  const { valid, errors } = validateRequest(body);

  if (!valid) {
    return Response.json(
      { errors },
      { status: 400 }
    );
  }

  // Proceed...
}

Logging Errors

// Always log errors for debugging
export async function POST(request) {
  try {
    const result = await processRequest(request);
    return Response.json(result);

  } catch (error) {
    // Log the full error internally
    console.error('ERROR in POST /api/auth/login:', {
      message: error.message,
      stack: error.stack,
      timestamp: new Date()
    });

    // Return generic error to user
    return Response.json(
      { error: 'Something went wrong' },
      { status: 500 }
    );
  }
}

Response Structure

Consistent error responses help frontend developers:

// Success
Response.json({
  success: true,
  data: { ... }
})

// Error
Response.json({
  success: false,
  error: 'What went wrong',
  errorCode: 'INVALID_EMAIL'  // Machine-readable
}, { status: 400 })

Key Concepts

  • Validation: Check data is correct
  • 400: Bad Request (validation failed)
  • 401: Unauthorized (not logged in)
  • 403: Forbidden (no permission)
  • 500: Server Error (unexpected failure)
  • Try/Catch: Catch unexpected errors
